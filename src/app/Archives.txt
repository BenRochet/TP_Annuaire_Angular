import { Component, OnInit, signal } from '@angular/core';
import { firstValueFrom } from 'rxjs';
import { Service, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';


Personne {
  gender : string,
  name :{
    title :string,
    first: string,
    last: string,
  },
  location : {
    street : {
      number: number,
      name : string,
    },
    city : string,
    state : string,
    country : string,
    postCode : number,
  },
  login : {
    username : string,
    password : string,
  },
 };


@Component({
  imports: [],
  selector: 'app-api-service',
  styleUrl: './api-service.css',
  templateUrl: './api-service.html',
})


export class APIService implements OnInit{

  private readonly url = 'https://randomuser.me/api/?seed=cda2&results=30&nat=fr';
  private readonly http = inject(HttpClient);

  protected readonly personnes = signal<Personne[]>([]);
 
  async ngOnInit() {
    this.personnes.set(await this.getAll());
  }
 
  async getAll(): Promise<Personne[]> {
  const response = await firstValueFrom(this.http.get<{ results: Personne[] }>(this.url));
  return response.results;
  }
}
